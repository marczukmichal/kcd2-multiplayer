@code {
    // 1. MODEL DANYCH (To co przyjdzie z API)
    public class ServerInfo
    {
        public string Name { get; set; } = "";
        public string Ip { get; set; } = "";
        public int Port { get; set; }
        public string MapName { get; set; } = "";
        public int Players { get; set; }
        public int MaxPlayers { get; set; }
        public int Ping { get; set; }
    }

    private string errorMessage = "";

    // 2. SZTUCZNE DANE (Na razie wpisane na sztywno)
    private List<ServerInfo> servers = new List<ServerInfo>
    {
        new ServerInfo { Name = "Oficjalny Serwer PL #1", Ip = "127.0.0.1", Port = 27015, MapName = "skalitz_survival", Players = 12, MaxPlayers = 64, Ping = 25 },
        new ServerInfo { Name = "Roleplay Tavern", Ip = "192.168.1.5", Port = 7777, MapName = "rataje_city", Players = 60, MaxPlayers = 64, Ping = 42 },
        new ServerInfo { Name = "Dev Test Server", Ip = "10.0.0.2", Port = 28000, MapName = "debug_map", Players = 1, MaxPlayers = 10, Ping = 5 },
    };

    // 3. LOGIKA KOLOROWANIA PINGU
    private string GetPingColor(int ping)
    {
        if (ping < 50) return "#00ff00"; // Zielony
        if (ping < 100) return "#ffff00"; // Żółty
        return "#ff0000"; // Czerwony
    }

    // 4. URUCHAMIANIE GRY
    private void LaunchGame(ServerInfo server)
    {
        errorMessage = ""; // Czyść błędy
        try
        {
            // === KONFIGURACJA ŚCIEŻKI ===
            // Dla testu ustawiamy NOTATNIK. Jak klikniesz "GRAJ", odpali się notatnik.
            // Docelowo zmień to na: @"C:\Program Files (x86)\Steam\steamapps\common\KingdomComeDeliverance\Bin\Win64\KingdomCome.exe"
            string gameExePath = "notepad.exe";

            // Weryfikacja czy plik istnieje (dla notatnika pomijamy)
            if (gameExePath != "notepad.exe" && !File.Exists(gameExePath))
            {
                errorMessage = $"Nie znaleziono pliku gry: {gameExePath}. Sprawdź ścieżkę w kodzie.";
                return;
            }

            // Budowanie argumentów
            // Wynik: +map nazwa_mapy --kcdmp-ip 1.2.3.4 --kcdmp-port 1234
            string args = $"+map {server.MapName} --kcdmp-ip {server.Ip} --kcdmp-port {server.Port}";

            var startInfo = new ProcessStartInfo
                {
                    FileName = gameExePath,
                    Arguments = args,
                    UseShellExecute = false,
                // WorkingDirectory jest ważne dla gier, żeby ładowały assety
                // WorkingDirectory = Path.GetDirectoryName(gameExePath)
                };

            Process.Start(startInfo);
        }
        catch (Exception ex)
        {
            errorMessage = $"Błąd krytyczny: {ex.Message}";
        }
    }
